## 2. WebGL截图实现 (v5.0核心功能)

### 2.1 技术原理

**WebGL Canvas截图流程：**
1. **WebGL上下文拦截** (`map.js:interceptWebGLContext`)
   - 拦截HTMLCanvasElement.prototype.getContext
   - 强制设置 `preserveDrawingBuffer: true`
   - 确保WebGL缓冲区在渲染后保留，可供读取

2. **像素读取** (`screenshot.js:captureWebGLCanvas`)
   - 使用 `gl.readPixels()` 读取WebGL像素数据
   - WebGL坐标系上下翻转，需要修正
   - 检查像素数据有效性（避免全黑/全透明）

3. **Canvas绘制水印** (`screenshot.js:addWatermark`)
   - 预加载车辆logo (`preloadLogo`)
   - 绘制中心点标记 (`drawCenterMarker`)
   - 绘制底部水印信息 (`addWatermarkToCanvas`)

### 2.2 中心点标记绘制

**标记设计规范：**
- 外圈发光效果：径向渐变，品牌颜色透明度0.3→0
- 阴影效果：`rgba(0,0,0,0.4)`，模糊12px
- 主标记圆圈：品牌颜色，直径60px
- 白色边框：4px宽度
- 车辆logo：28x28px，居中显示
  - 深色背景：应用 `brightness(0) invert(1)` 滤镜转白色
  - 浅色背景：保持原色
- 车辆名称：标记下方10px，品牌颜色文字

### 2.3 水印信息布局

**底部80px高度水印条：**
- 左对齐内容（x=20）：
  - 🚗 车型名称（y=-50, 20px粗体）
  - 📍 车牌号（y=-25, 16px）
  - 📍 地址（y=-8, 14px，最多40字符）

- 右对齐内容（x=width-20）：
  - 🕒 停车时间（y=-50, 14px）
  - 🔍 放大倍数（y=-25, 14px）

### 2.4 关键代码

**WebGL上下文拦截：**
```javascript
interceptWebGLContext() {
    const originalGetContext = HTMLCanvasElement.prototype.getContext;
    HTMLCanvasElement.prototype.getContext = function(contextType, ...args) {
        if (contextType === 'webgl' || contextType === 'experimental-webgl') {
            let contextAttributes = args[0] || {};
            contextAttributes.preserveDrawingBuffer = true;  // 关键！
            contextAttributes.premultipliedAlpha = false;
            contextAttributes.alpha = true;
            return originalGetContext.call(this, contextType, contextAttributes);
        }
        return originalGetContext.call(this, contextType, ...args);
    };
}
```

**像素读取和翻转：**
```javascript
const pixels = new Uint8Array(width * height * 4);
gl.readPixels(0, 0, width, height, gl.RGBA, gl.UNSIGNED_BYTE, pixels);

// WebGL像素上下翻转，需要翻转回来
const flippedData = new Uint8Array(pixels.length);
for (let y = 0; y < height; y++) {
    for (let x = 0; x < width; x++) {
        const srcOffset = ((height - y - 1) * width + x) * 4;
        const destOffset = (y * width + x) * 4;
        flippedData[destOffset] = pixels[srcOffset];
        flippedData[destOffset + 1] = pixels[srcOffset + 1];
        flippedData[destOffset + 2] = pixels[srcOffset + 2];
        flippedData[destOffset + 3] = pixels[srcOffset + 3];
    }
}
```

```javascript
// vehicle-card.js
class VehicleCard {
    constructor(vehicle) {
        this.vehicle = vehicle;
    }

    render() {
        const card = document.createElement('div');
        card.className = 'col-6';
        card.innerHTML = `
            <div class="vehicle-card" data-car-id="${this.vehicle.id}">
                <div class="vehicle-logo">
                    <img src="${this.vehicle.logo}" alt="${this.vehicle.brand}">
                </div>
                <div class="vehicle-name">${this.vehicle.name}</div>
                <div class="vehicle-plate">${this.vehicle.plate}</div>
            </div>
        `;

        // 添加点击事件
        card.querySelector('.vehicle-card').addEventListener('click', () => {
            this.selectVehicle();
        });

        return card;
    }

    selectVehicle() {
        window.location.href = `map-page.html?car=${this.vehicle.id}`;
    }
}
```

```javascript
        const day = String(date.getDate()).padStart(2, '0');

    calculateDistance: function(lat1, lng1, lat2, lng2) {
        const R = 6371e3; // 地球半径（米）
        const φ1 = lat1 * Math.PI / 180;
        const φ2 = lat2 * Math.PI / 180;
        const Δφ = (lat2 - lat1) * Math.PI / 180;
        const Δλ = (lng2 - lng1) * Math.PI / 180;

        const a = Math.sin(Δφ/2) * Math.sin(Δφ/2) +
                  Math.cos(φ1) * Math.cos(φ2) *
                  Math.sin(Δλ/2) * Math.sin(Δλ/2);
        const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a));

        return R * c;
```

```javascript
// 定位相关功能（v2.0增强 - 支持默认位置回退）
class LocationManager {
    constructor() {
        this.currentPosition = null;
        this.watchId = null;
        this.callbacks = [];
        this.tencentLocationEnabled = false;
        this.defaultLocationUsed = false;
    }

    // 启用腾讯定位SDK
    enableTencentLocation() {
        if (typeof window.TencentLocation !== 'undefined') {
            this.tencentLocationEnabled = true;
            return true;
        }
        return false;
    }

    // 获取当前位置（支持默认位置回退）
    async getCurrentPosition(options = {}) {
        try {
            // 获取浏览器定位策略
            const strategy = browserDetection.getLocationStrategy();

            // 首先尝试浏览器定位
            const location = await this._tryBrowserLocation(strategy);
            this.defaultLocationUsed = false;
            return location;

        } catch (error) {
            console.warn('浏览器定位失败，使用默认位置:', error);
            // 定位失败时使用默认位置
            return this._createDefaultLocation();
        }
    }

    // 尝试浏览器定位
    _tryBrowserLocation(strategy) {
        return new Promise((resolve, reject) => {
            if (!navigator.geolocation) {
                reject(new Error('您的浏览器不支持定位功能'));
                return;
            }

            const options = {
                enableHighAccuracy: strategy.enableHighAccuracy,
                timeout: strategy.timeout,
                maximumAge: strategy.maximumAge
            };

            navigator.geolocation.getCurrentPosition(
                (position) => {
                    const location = {
                        lat: position.coords.latitude,
                        lng: position.coords.longitude,
                        accuracy: position.coords.accuracy,
                        altitude: position.coords.altitude,
                        altitudeAccuracy: position.coords.altitudeAccuracy,
                        heading: position.coords.heading,
                        speed: position.coords.speed,
                        timestamp: position.timestamp,
                        provider: 'browser',
                        browserStrategy: `${browserDetection.detect().isXiaomi ? '小米' : browserDetection.detect().isUC ? 'UC' : '标准'}浏览器`,
                        isDefaultLocation: false
                    };

                    this.currentPosition = location;
                    this.notifyCallbacks(location);
                    resolve(location);
                },
                (error) => {
                    // 处理定位错误，判断是否使用默认位置
                    const errorResult = this._handleGeolocationError(error);
                    if (errorResult.shouldUseDefaultLocation) {
                        reject(new Error('定位失败，将使用默认位置'));
                    } else {
                        reject(new Error(errorResult.message));
                    }
                },
                options
            );
        });
    }

    // 处理定位错误
    _handleGeolocationError(error) {
        let errorMessage;
        let shouldUseDefaultLocation = false;

        switch(error.code) {
            case error.PERMISSION_DENIED:
                errorMessage = ERROR_MESSAGES.PERMISSION_DENIED;
                shouldUseDefaultLocation = true; // 权限拒绝时使用默认位置
                break;
            case error.TIMEOUT:
                errorMessage = ERROR_MESSAGES.TIMEOUT;
                shouldUseDefaultLocation = true; // 超时时使用默认位置
                break;
            case error.POSITION_UNAVAILABLE:
                errorMessage = ERROR_MESSAGES.POSITION_UNAVAILABLE;
                shouldUseDefaultLocation = true; // 位置不可用时使用默认位置
                break;
            default:
                errorMessage = '获取位置失败：' + error.message;
                shouldUseDefaultLocation = true;
        }

        return {
            message: errorMessage,
            shouldUseDefaultLocation: shouldUseDefaultLocation
        };
    }

    // 创建默认位置
    _createDefaultLocation() {
        const defaultLocation = JSON.parse(JSON.stringify(MAP_CONFIG.defaultLocation));
        defaultLocation.timestamp = Date.now();
        defaultLocation.provider = '默认位置';
        defaultLocation.isDefaultLocation = true;
        defaultLocation.accuracy = 50; // 设置合理的默认精度

        console.log('使用默认位置:', defaultLocation);
        this.defaultLocationUsed = true;
        return defaultLocation;
    }

    // 监听位置变化
    watchPosition(callback) {
        if (!navigator.geolocation) {
            throw new Error('您的浏览器不支持定位功能');
        }

        this.callbacks.push(callback);

        if (this.watchId === null) {
            const options = {
                enableHighAccuracy: true,
                timeout: APP_CONFIG.locationTimeout,
                maximumAge: 5000
            };

            this.watchId = navigator.geolocation.watchPosition(
                (position) => {
                    const location = {
                        lat: position.coords.latitude,
                        lng: position.coords.longitude,
                        accuracy: position.coords.accuracy,
                        timestamp: position.timestamp
                    };

                    this.currentPosition = location;
                    this.notifyCallbacks(location);
                },
                (error) => {
                    console.error('定位监听错误:', error);
                },
                options
            );
        }
    }

    // 停止监听位置变化
    stopWatching() {
        if (this.watchId !== null) {
            navigator.geolocation.clearWatch(this.watchId);
            this.watchId = null;
        }
        this.callbacks = [];
    }

    // 通知所有回调
    notifyCallbacks(location) {
        this.callbacks.forEach(callback => {
            try {
                callback(location);
            } catch (error) {
                console.error('定位回调错误:', error);
            }
        });
    }

    // 获取定位精度描述
    getAccuracyDescription(accuracy) {
        if (accuracy < 10) {
            return '极高精度';
        } else if (accuracy < 50) {
            return '高精度';
        } else if (accuracy < 100) {
            return '中等精度';
        } else if (accuracy < 500) {
            return '低精度';
        } else {
            return '极低精度';
        }
    }

    // 检查定位权限
    checkLocationPermission() {
        return new Promise((resolve) => {
            if ('permissions' in navigator) {
                navigator.permissions.query({ name: 'geolocation' }).then((result) => {
                    resolve(result.state);
                }).catch(() => {
                    resolve('unknown');
                });
            } else {
                resolve('unknown');
            }
        });
    }

    // 请求定位权限
    requestLocationPermission() {
        return this.getCurrentPosition().catch(() => {
            // 忽略错误，只是为了触发权限请求
        });
    }
}
```

```javascript
            }
            return cachedUrl.url;
        }

        try {
            // 生成腾讯地图分享页面
            const sharePageUrl = await this._generateTencentSharePage(location, carInfo);

            const cacheData = {
                url: sharePageUrl,
                timestamp: Date.now(),
                source: '腾讯地图分享页面'
            };

            Utils.storage.set(`sharepage_${cacheKey}`, cacheData, CACHE_CONFIG.screenshotCacheTime);

            console.log('✅ 腾讯地图分享页面生成成功');
            return sharePageUrl;

        } catch (error) {
            console.error('❌ 分享页面生成失败:', error.message);
            throw new Error('分享页面生成失败：' + error.message);
        }
    }

    // 生成腾讯地图标记页面URL（用于iframe嵌入）
    _generateTencentMarkerUrl(location, carInfo, address) {
        // 使用腾讯地图API生成标记页面
        const params = new URLSearchParams({
            marker: `coord:${location.lat},${location.lng};title:${carInfo.name};addr:${address}`,
            key: MAP_CONFIG.key,
            referer: window.location.hostname
        });

        // 使用腾讯地图URI API，这个可以在iframe中显示
        return `https://apis.map.qq.com/uri/v1/marker?${params.toString()}`;
    }

    // 逆地理编码获取地址
    async getAddressFromLocation(location) {
        if (!location) {
            return '位置未知';
        }

        const params = new URLSearchParams({
            location: `${location.lat},${location.lng}`,
            key: MAP_CONFIG.key,
            get_poi: 1
        });

        const url = `https://apis.map.qq.com/ws/geocoder/v1/?${params.toString()}`;

        try {
            const response = await fetch(url);
            const data = await response.json();

            if (data.status === 0 && data.result) {
                return data.result.address || '位置未知';
            } else {
                throw new Error(data.message || '地址解析失败');
            }
        } catch (error) {
            console.error('地址解析失败:', error);
            return '地址解析失败';
        }
    }

    // 销毁地图
    destroy() {
        if (this.marker) {
            this.marker.setMap(null);
            this.marker = null;
        }

        if (this.map) {
            this.map.destroy();
            this.map = null;
        }

        this.isMapLoaded = false;
    }
}
```

```javascript
// 主应用类
class VehicleLocationApp {
    constructor() {

    // 初始化应用

        try {
            // 解析URL参数
            const params = Utils.getUrlParams();
            const carId = params.car;

            if (!carId || !VEHICLE_CONFIG[carId]) {
                throw new Error(ERROR_MESSAGES.INVALID_QRCODE);

            this.initUI();

            this.showPage('loading-page');

            // 绑定事件
            this.bindEvents();

            this.isInitialized = true;

        } catch (error) {
            this.handleError(error);
        }
    }

    // 初始化UI
    initUI() {
        // 设置车辆信息
        document.getElementById('car-name').textContent = this.currentCar.name;
        document.getElementById('car-plate').textContent = this.currentCar.plate;
        document.title = `${this.currentCar.name} - ${APP_CONFIG.title}`;
    }

    // 绑定事件
    bindEvents() {
        // 刷新位置按钮
        document.getElementById('refresh-btn').addEventListener('click', () => {
            this.refreshLocation();
        });

        // 分享位置按钮
        document.getElementById('share-btn').addEventListener('click', () => {
            this.showSharePage();
        });

        // 复制位置信息按钮
        document.getElementById('copy-btn').addEventListener('click', () => {

        });

        // 保存图片按钮

            this.saveImage();
        });

        // 错误模态框确认按钮
        document.getElementById('error-ok-btn').addEventListener('click', () => {
            this.hideErrorModal();
        });

        // 页面可见性变化
        document.addEventListener('visibilitychange', () => {
            if (!document.hidden && this.isInitialized) {
                this.refreshLocation();
            }
        });
    }

    // 开始定位追踪
    async startLocationTracking() {
        try {
            // 获取当前位置
            const location = await this.locationManager.getCurrentPosition();
            this.currentLocation = location;

            // 初始化地图
            await this.initializeMap();

            // 显示地图页面
            this.showMapPage();

            // 开始监听位置变化
            this.locationManager.watchPosition((newLocation) => {
                this.onLocationUpdate(newLocation);
            });

        } catch (error) {
            this.handleLocationError(error);
        }
    }

    // 初始化地图
    async initializeMap() {
        const container = document.getElementById('map-container');
        await this.mapManager.initializeMap(container, this.currentLocation);
        await this.updateMapDisplay();
    }

    // 更新地图显示（v2.0增强 - 含经纬度显示）
    async updateMapDisplay() {
        if (!this.currentLocation || !this.mapManager.isMapLoaded) {
            return;
        }

        // 更新标记
        this.mapManager.addMarker(this.currentLocation, this.currentCar);

        // 更新地址信息
        const address = await this.mapManager.getAddressFromLocation(this.currentLocation);
        this.updateAddressDisplay(address);

        // 更新经纬度信息显示
        this.updateCoordinatesDisplay(this.currentLocation);
    }

    // 更新经纬度信息显示（v2.0新增）
    updateCoordinatesDisplay(location) {
        if (!location) {
            return;
        }

        try {
            // 显示经纬度信息容器
            const coordinatesInfo = document.getElementById('coordinates-info');
            if (coordinatesInfo) {
                coordinatesInfo.style.display = 'block';
            }

            // 更新经纬度显示
            const coordinatesDisplay = document.getElementById('coordinates-display');
            if (coordinatesDisplay) {
                const lat = location.lat.toFixed(6);
                const lng = location.lng.toFixed(6);
                coordinatesDisplay.textContent = `${lat}, ${lng}`;

                // 如果是默认位置，添加标识

                    coordinatesDisplay.style.color = '#ff9500';
                    coordinatesDisplay.style.fontStyle = 'italic';
                } else {
                    coordinatesDisplay.style.color = '#0066cc';
                    coordinatesDisplay.style.fontStyle = 'normal';
                }
            }

            // 更新精度信息
            const accuracyDisplay = document.getElementById('accuracy-display');
            if (accuracyDisplay) {
                const accuracy = location.accuracy ? Math.round(location.accuracy) : '未知';
                if (location.isDefaultLocation) {
                    accuracyDisplay.innerHTML = `${accuracy}米 <small style="color: #ff9500;">(测试位置)</small>`;
                } else {
                    accuracyDisplay.textContent = `${accuracy}米`;
```
